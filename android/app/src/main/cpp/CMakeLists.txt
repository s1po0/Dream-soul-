cmake_minimum_required(VERSION 3.22.1)
project(dreamsoul_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
# Optimize the CPU engine even in a debug-signed preview APK.
if(ANDROID)
  set(CMAKE_C_FLAGS_DEBUG "-O3 -g0")
  set(CMAKE_CXX_FLAGS_DEBUG "-O3 -g0")
endif()
# Android x86_64 does not guarantee AVX; keep emulator/older Intel builds portable.
if(ANDROID_ABI STREQUAL "x86_64")
  set(GGML_AVX OFF CACHE BOOL "" FORCE)
  set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
  set(GGML_FMA OFF CACHE BOOL "" FORCE)
  set(GGML_F16C OFF CACHE BOOL "" FORCE)
  set(GGML_BMI2 OFF CACHE BOOL "" FORCE)
endif()
include(FetchContent)
FetchContent_Declare(llama
  URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/620d2225cc436ab0b001e3200e590e3b286ef9ab
  URL_HASH SHA256=1d4a987d9bd048f6e93107e838026438df70fbde237fc0c69c63d600512b4bd9)
FetchContent_MakeAvailable(llama)
add_library(dreamsoul_llm SHARED offline.cpp)
target_link_libraries(dreamsoul_llm PRIVATE llama)
if(ANDROID)
  target_link_libraries(dreamsoul_llm PRIVATE android log)
  target_link_options(dreamsoul_llm PRIVATE "-Wl,-z,max-page-size=16384")
else()
  # Host compilation catches upstream API drift without needing the Android SDK.
  find_package(JNI REQUIRED)
  target_include_directories(dreamsoul_llm PRIVATE ${JNI_INCLUDE_DIRS})
endif()
